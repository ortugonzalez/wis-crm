const buckets = { secretary_tasks: [], prospects: [], sales_campaigns: [], secretary_preferences: [] };
for (const item of items) {
  const source = item.json._source;
  if (source && buckets[source]) buckets[source].push(item.json);
}

const { secretary_tasks, prospects, sales_campaigns, secretary_preferences } = buckets;
const prefs = secretary_preferences[0] ?? {
  display_name: 'Ortu',
  mode: 'normal',
  reminder_minutes: 45,
  max_attempts: 2,
  current_focus: '',
  preferred_segments: [],
  energy_mode: 'equilibrado',
  current_state: 'arrancando',
  daily_contact_target: 10,
  daily_followup_target: 3,
  daily_proposal_target: 1,
  relationship_goal: '',
  one_thing_rule: true,
  coaching_style: 'concreto',
};

if (prefs.mode === 'pausa') return [];

const now = new Date();
const today = now.toISOString().slice(0, 10);
const firstName = String(prefs.display_name || 'Ortu').split(' ')[0];
const focus = prefs.current_focus ? String(prefs.current_focus).trim() : '';
const preferredSegments = Array.isArray(prefs.preferred_segments) ? prefs.preferred_segments.filter(Boolean) : [];
const energyMode = String(prefs.energy_mode || 'equilibrado');
const currentState = String(prefs.current_state || 'arrancando');
const coachingStyle = String(prefs.coaching_style || 'concreto');
const contactTarget = Number(prefs.daily_contact_target || 10);
const followTarget = Number(prefs.daily_followup_target || 3);
const proposalTarget = Number(prefs.daily_proposal_target || 1);
const reminderMinutes = Number(prefs.reminder_minutes || 45);
const maxAttempts = Number(prefs.max_attempts || 2);
const oneThingRule = Boolean(prefs.one_thing_rule ?? true);

function daysSince(dateLike) {
  if (!dateLike) return 999;
  return Math.floor((Date.now() - new Date(dateLike).getTime()) / (1000 * 60 * 60 * 24));
}

function safeCompany(p) {
  return p.company || p.name || 'este contacto';
}

function warmIntro() {
  if (coachingStyle === 'amable') return `${firstName}, vamos simple.`;
  if (coachingStyle === 'exigente') return `${firstName}, foco.`;
  return `${firstName}, una cosa por vez.`;
}

function compactInstruction(text) {
  if (!oneThingRule) return text;
  return `${text}\n\nSolo resolvamos esto ahora.`;
}

function contactAsk() {
  if (energyMode === 'suave' || currentState === 'saturado') return 1;
  if (prefs.mode === 'liviano') return Math.min(3, Math.max(1, contactTarget));
  if (prefs.mode === 'sprint') return Math.max(3, Math.min(contactTarget, 10));
  return Math.max(2, Math.min(contactTarget, 6));
}

function followAsk() {
  if (energyMode === 'suave' || currentState === 'saturado') return 1;
  if (prefs.mode === 'liviano') return Math.min(2, Math.max(1, followTarget));
  if (prefs.mode === 'sprint') return Math.max(2, Math.min(followTarget, 5));
  return Math.max(1, Math.min(followTarget, 3));
}

function proposalAsk() {
  if (energyMode === 'suave' || currentState === 'saturado') return Math.min(1, proposalTarget);
  return Math.max(1, Math.min(proposalTarget, 2));
}

const commitmentTask = secretary_tasks.find(t =>
  t.task_type === 'commitment_check' &&
  t.task_date === today &&
  ['pendiente', 'en_progreso'].includes(t.status)
);

if (commitmentTask) {
  const minSince = Math.floor((now - new Date(commitmentTask.reminded_at || commitmentTask.created_at)) / 60000);
  if (minSince < reminderMinutes) return [];
  return [{
    json: {
      mode: 'remind',
      task_id: commitmentTask.id,
      attempts: Number(commitmentTask.attempts || 0) + 1,
      reminded_at: now.toISOString(),
      message: `${warmIntro()} Ayer quedamos en esto:\n"${commitmentTask.instruction}"\n\nConfirmame solo estado: hecho, despues o bloqueo.`
    }
  }];
}

const last7 = secretary_tasks.filter(t => {
  if (t.task_type === 'commitment_check') return false;
  const d = new Date(t.task_date || t.created_at);
  return (now - d) / (1000 * 60 * 60 * 24) <= 7;
});
const done7 = last7.filter(t => t.status === 'hecha').length;
const rate7 = last7.length >= 4 ? Math.round(done7 / last7.length * 100) : null;
const recentTypes = last7.slice(-4).map(t => t.task_type);
const repeatedType = recentTypes.length >= 3 && new Set(recentTypes).size === 1 ? recentTypes[0] : null;

const birthdays = prospects
  .filter(p => p.birthday)
  .map(p => {
    const b = new Date(p.birthday);
    const next = new Date(now.getFullYear(), b.getMonth(), b.getDate());
    return { prospect: p, days: Math.floor((next.getTime() - now.getTime()) / (1000 * 60 * 60 * 24)) };
  })
  .filter(item => item.days >= 0 && item.days <= 7)
  .sort((a, b) => a.days - b.days);

const actionableProspects = prospects
  .filter(p => p.decision_status !== 'descartar')
  .filter(p => !p.paused_until || new Date(p.paused_until).getTime() <= now.getTime())
  .map(p => ({ prospect: p, days: daysSince(p.last_contact_at || p.updated_at || p.created_at) }))
  .sort((a, b) => {
    const score = item => {
      const temp = item.prospect.temperature === 'caliente' ? 5 : item.prospect.temperature === 'tibio' ? 3 : 1;
      const stage = ['propuesta', 'reunion', 'contactado', 'frio', 'cliente'].indexOf(item.prospect.stage);
      const decision = item.prospect.decision_status === 'avanzar' ? 2 : item.prospect.decision_status === 'nutrir' ? 1 : 0;
      return temp + (5 - stage) + decision;
    };
    return score(b) - score(a);
  });

const hotFollowUps = actionableProspects.filter(item => item.prospect.temperature === 'caliente' || (['propuesta', 'reunion', 'contactado'].includes(item.prospect.stage) && item.days >= 3));
const nurtureProspects = actionableProspects.filter(item => item.prospect.decision_status === 'nutrir' && item.days >= 5);
const staleNonClient = actionableProspects.filter(item => item.prospect.stage !== 'cliente' && item.days >= 7);
const campaignActiva = sales_campaigns.find(c => c.status === 'activa');

function buildNextTask() {
  if (birthdays[0] && prefs.relationship_goal) {
    const b = birthdays[0];
    return {
      task_type: 'relationship_touch',
      title: `Relacion: ${safeCompany(b.prospect)}`,
      instruction: compactInstruction(`Cumple en ${b.days} dia(s). Escribile algo humano a ${b.prospect.name}${b.prospect.company ? ` de ${b.prospect.company}` : ''}.`),
      expected_response: 'Mensaje enviado o texto listo.',
      priority: 'media',
    };
  }

  if (hotFollowUps[0]) {
    const item = hotFollowUps[0];
    return {
      task_type: 'follow_up',
      title: `Seguimiento prioritario: ${safeCompany(item.prospect)}`,
      instruction: compactInstruction(`${safeCompany(item.prospect)} esta ${item.prospect.temperature} y en ${item.prospect.stage}. Hacelo avanzar hoy con proximo paso concreto.`),
      expected_response: 'Accion hecha o bloqueo exacto.',
      priority: item.prospect.stage === 'propuesta' ? 'alta' : 'media',
    };
  }

  if (nurtureProspects[0]) {
    const item = nurtureProspects[0];
    return {
      task_type: 'relationship_touch',
      title: `Nutrir: ${safeCompany(item.prospect)}`,
      instruction: compactInstruction(`Retoma a ${item.prospect.name} con un mensaje corto. Objetivo: definir si sigue en radar o si queda pausado.`),
      expected_response: 'Mensaje enviado o decision tomada.',
      priority: 'media',
    };
  }

  if (prospects.length < 10) {
    const amount = contactAsk();
    return {
      task_type: 'collect_contacts',
      title: 'Nutrir base comercial',
      instruction: compactInstruction(`Mandame ${amount} contacto${amount === 1 ? '' : 's'} nuevo${amount === 1 ? '' : 's'}${focus ? ` para ${focus}` : ''}. Formato libre: nombre, empresa y rubro.`),
      expected_response: 'Al menos un contacto nuevo.',
      priority: 'media',
    };
  }

  if (!campaignActiva) {
    const segmentHint = preferredSegments.length ? preferredSegments.join(', ') : 'el segmento mas urgente';
    return {
      task_type: 'create_campaign',
      title: 'Definir campana de la semana',
      instruction: compactInstruction(`Elegi un frente para ordenar el dia. Opciones sugeridas: ${segmentHint}.`),
      expected_response: 'Segmento elegido.',
      priority: 'media',
    };
  }

  if (staleNonClient[0]) {
    const item = staleNonClient[0];
    return {
      task_type: 'follow_up',
      title: `Reactivar: ${safeCompany(item.prospect)}`,
      instruction: compactInstruction(`Resolve ${followAsk()} seguimiento(s) concretos empezando por ${item.prospect.name}. Nada generico: mensaje, propuesta o siguiente paso.`),
      expected_response: 'Accion hecha o bloqueo.',
      priority: 'media',
    };
  }

  return {
    task_type: 'execution_block',
    title: `Bloque comercial: ${campaignActiva.name}`,
    instruction: compactInstruction(`Objetivo del bloque: ${contactAsk()} contacto(s), ${followAsk()} follow-up(s) y ${proposalAsk()} propuesta(s) como maximo.`),
    expected_response: 'Progreso del bloque.',
    priority: 'baja',
  };
}

const nextTask = buildNextTask();
const patternAlert = repeatedType === 'collect_contacts'
  ? '\n\nCuidado: venis cargando contactos seguido. La proxima ficha deberia ser outreach real.'
  : rate7 !== null && rate7 < 40
    ? `\n\nDato util: esta semana completaste ${done7}/${last7.length} misiones. Conviene achicar antes que acumular.`
    : '';

const active = secretary_tasks.find(t =>
  ['pendiente', 'en_progreso'].includes(t.status) &&
  t.task_type !== 'commitment_check'
);

if (active) {
  const minSince = Math.floor((now - new Date(active.reminded_at || active.created_at)) / 60000);
  if (minSince < reminderMinutes) return [];

  const attempts = Number(active.attempts || 0);
  if (attempts >= maxAttempts) {
    return [{
      json: {
        mode: 'auto_complete',
        task_id: active.id,
        message: `${warmIntro()} cierro esta mision por inercia y te dejo una mas clara.\n\n${nextTask.title}\n${nextTask.instruction}${patternAlert}\n\nResponde: hecho, despues o mandame el dato.`,
        ...nextTask,
        task_date: today,
        reminded_at: now.toISOString(),
      }
    }];
  }

  let msg = `${warmIntro()} sigue pendiente:\n${active.title}\n\n${active.instruction}`;
  if (attempts >= 1) {
    if (active.task_type === 'collect_contacts') {
      msg = `${warmIntro()} te la simplifico: pasame solo 1 contacto ahora.`;
    } else if (active.task_type === 'follow_up') {
      msg = `${warmIntro()} version corta: move esta ficha y contame solo resultado o bloqueo.`;
    } else if (active.task_type === 'create_campaign') {
      msg = `${warmIntro()} elegi uno y listo: ${preferredSegments.join(', ') || 'hoteles, logistica o retail'}.`;
    } else {
      msg = `${warmIntro()} cerra esta mision primero.`;
    }
  }

  return [{
    json: {
      mode: 'remind',
      task_id: active.id,
      attempts: attempts + 1,
      reminded_at: now.toISOString(),
      message: `${msg}${patternAlert}\n\nResponde: hecho, despues o mandame el dato.`,
    }
  }];
}

return [{
  json: {
    mode: 'new_task',
    ...nextTask,
    task_date: today,
    reminded_at: now.toISOString(),
    message: `${warmIntro()}\n\n${nextTask.title}\n${nextTask.instruction}${patternAlert}\n\nResponde: hecho, despues o mandame el dato.`,
  }
}];
